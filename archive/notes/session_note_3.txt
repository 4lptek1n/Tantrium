Başlama yöntemi şu. En hızlı ve en temiz motor EGF katsayı çıkarımı:

G_\lambda(z,u)
=
\exp\left(
\frac{uz}{1-\lambda u}
-\frac{u^2}{4(1-\lambda u)}
-\frac{u^2}{48}\big((1-\lambda u)^{-2}-1\big)
\right)

P_{\lambda,d}(z)=d![u^d]G_\lambda(z,u).

SymPy kodu:

import sympy as sp
z, u, lam, t = sp.symbols("z u lam t")
def P(d):
    S = (
        u*z/(1-lam*u)
        - u**2/(4*(1-lam*u))
        - u**2/sp.Integer(48)*((1-lam*u)**(-2)-1)
    )
    G = sp.series(sp.exp(S), u, 0, d+1).removeO()
    return sp.expand(sp.factorial(d) * G.coeff(u, d))
def monic(poly):
    poly = sp.Poly(sp.expand(poly), z)
    return sp.expand(poly.as_expr() / poly.LC())
def sturm_pivots(d):
    F = []
    pivots = []
    F.append(monic(P(d)))
    F.append(monic(sp.diff(F[0], z)))
    for j in range(1, d):
        A = sp.Poly(F[j-1], z)
        B = sp.Poly(F[j], z)
        q, r = sp.div(A, B, domain="QQ[lam]")
        r = -sp.expand(r.as_expr())
        if r == 0:
            break
        R = sp.Poly(r, z)
        rho = sp.simplify(R.LC())
        pivots.append(sp.factor(rho))
        F.append(monic(r))
    return F, pivots
def pivot_data(d):
    F, pivots = sturm_pivots(d)
    for i, rho in enumerate(pivots, start=1):
        print(f"rho_{d},{i} =")
        print(sp.factor(rho.subs(lam**2, t)))
        print()
# örnek
pivot_data(6)

def even_in_t(expr):
    expr = sp.expand(expr)
    expr = expr.subs(lam**2, t)
    return sp.factor(expr)
def normalize_poly(poly_t):
    poly_t = sp.Poly(sp.factor(poly_t), t)
    c0 = poly_t.coeff_monomial(t**0)
    return sp.factor(poly_t.as_expr() / c0)

Katsayı interpolasyonu:

def interpolate_coeffs(values, shift):
    # values: [(d, poly_in_t), ...]
    polys = []
    maxdeg = max(sp.Poly(p, t).degree() for _, p in values)
    n = sp.symbols("n")
    for k in range(maxdeg + 1):
        pts = []
        for d, p in values:
            coeff = sp.Poly(p, t).coeff_monomial(t**k)
            pts.append((d - shift, coeff))
        poly_n = sp.interpolate(pts, n)
        polys.append(sp.factor(poly_n))
    return polys

Kısaca Yusuf’a verilecek motor bu:

P_d=d![u^d]G_\lambda

→ normalize Sturm bölmesi
→ pivotları çıkar
→ bilinen H faktörlerini ayır
→ H_{d,j}/H_{d,j}(0) normalize et
→ d-(j+1)=n ile interpolate et
→ katsayıların pozitif polinom olduğunu kontrol et.
